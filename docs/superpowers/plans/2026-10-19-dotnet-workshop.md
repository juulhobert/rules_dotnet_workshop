```bash
#!/usr/bin/env bash
# next_exercise.sh - move to the next exercise of the workshop.
#
# Present identically on every branch. DESTRUCTIVE: discards all uncommitted
# changes and untracked files in the working tree before checking out the
# next branch. Committed work survives.
set -euo pipefail

SEQUENCE=(
    "main"
    "exercise-000-intro"
    "exercise-001-hello-bazel"
    "exercise-002-nuget-resolve"
    "exercise-003-gazelle"
    "exercise-004-change"
)

current=$(git rev-parse --abbrev-ref HEAD)

next=""
for i in "${!SEQUENCE[@]}"; do
    if [[ "${SEQUENCE[$i]}" == "$current" ]]; then
        next="${SEQUENCE[$((i + 1))]:-}"
        break
    fi
done

if [[ -z "$next" ]]; then
    if [[ "$current" == "${SEQUENCE[-1]}" ]]; then
        echo "Workshop complete! You have finished the last exercise."
        exit 0
    fi
    echo "ERROR: '$current' is not a workshop branch." >&2
    echo "Check out one of: ${SEQUENCE[*]}" >&2
    exit 1
fi

cat <<EOF
!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
WARNING: this script is DESTRUCTIVE.

All uncommitted changes and untracked files in this working tree will be
permanently lost (git clean -fdx + git reset --hard). Committed work is
safe: commit or 'git stash' anything you want to keep before continuing.
!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!

Next exercise: $next
EOF

answer=""
if ! read -r -p "Type 'yes' to continue: " answer; then
    answer=""
fi
if [[ "$answer" != "yes" ]]; then
    echo "Aborted."
    exit 1
fi

git clean -fdx
git reset --hard
git checkout "$next"

echo
echo "You are now on $next."
echo "Open README.md to start the exercise."
```

```csharp
using System;
using System.Collections.Generic;

namespace TicTacToe.Engine;

/// <summary>
/// A 3x3 tic-tac-toe board. Cells are numbered 0-8, left to right, top to bottom.
/// </summary>
public class Board
{
    private static readonly int[][] Lines =
    {
        new[] { 0, 1, 2 }, new[] { 3, 4, 5 }, new[] { 6, 7, 8 }, // rows
        new[] { 0, 3, 6 }, new[] { 1, 4, 7 }, new[] { 2, 5, 8 }, // columns
        new[] { 0, 4, 8 }, new[] { 2, 4, 6 }, // diagonals
    };

    private readonly Player?[] _cells = new Player?[9];

    /// <summary>Returns the mark in the given cell, or null when the cell is empty.</summary>
    public Player? GetCell(int index) => _cells[index];

    public bool IsEmpty(int index) => _cells[index] is null;

    /// <summary>The indices of all empty cells, in order.</summary>
    public IEnumerable<int> EmptyCells()
    {
        for (var i = 0; i < 9; i++)
        {
            if (_cells[i] is null)
            {
                yield return i;
            }
        }
    }

    /// <summary>Places a mark. Throws when the cell is already taken.</summary>
    public void Place(int index, Player player)
    {
        if (!IsEmpty(index))
        {
            throw new InvalidOperationException($"Cell {index} is already taken.");
        }

        _cells[index] = player;
    }

    /// <summary>The winning player, or null when there is no winner (yet).</summary>
    public Player? Winner()
    {
        foreach (var line in Lines)
        {
            var first = _cells[line[0]];
            if (first is not null && first == _cells[line[1]] && first == _cells[line[2]])
            {
                return first;
            }
        }

        return null;
    }

    public bool IsFull()
    {
        foreach (var cell in _cells)
        {
            if (cell is null)
            {
                return false;
            }
        }

        return true;
    }

    public bool IsGameOver() => Winner() is not null || IsFull();

    /// <summary>A copy of this board, used by the computer to try moves out.</summary>
    public Board Clone()
    {
        var copy = new Board();
        for (var i = 0; i < 9; i++)
        {
            if (_cells[i] is Player player)
            {
                copy.Place(i, player);
            }
        }

        return copy;
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace TicTacToe.Engine;

/// <summary>
/// Picks moves with a simple priority: take a winning move, block the
/// opponent's winning move, otherwise play a random empty cell.
/// </summary>
public class ComputerPlayer
{
    private readonly Random _random = new();
    private readonly Player _me;
    private readonly Player _opponent;

    public ComputerPlayer(Player me, Player opponent)
    {
        _me = me;
        _opponent = opponent;
    }

    public int ChooseMove(Board board)
    {
        return FindWinningMove(board, _me)
            ?? FindWinningMove(board, _opponent)
            ?? RandomMove(board);
    }

    /// <summary>The cell that wins the game for <paramref name="player"/>, if any.</summary>
    private int? FindWinningMove(Board board, Player player)
    {
        foreach (var cell in board.EmptyCells())
        {
            var copy = board.Clone();
            copy.Place(cell, player);
            if (copy.Winner() == player)
            {
                return cell;
            }
        }

        return null;
    }

    private int RandomMove(Board board)
    {
        var empty = new List<int>(board.EmptyCells());
        return empty[_random.Next(empty.Count)];
    }
}
```

```csharp
using System;
using TicTacToe.Engine;

var board = new Board();
var computer = new ComputerPlayer(Player.O, Player.X);

Console.WriteLine("Tic-Tac-Toe — you are X, the computer is O.");
Console.WriteLine("Cells are numbered 1-9, left to right, top to bottom.");
Console.WriteLine();

while (!board.IsGameOver())
{
    Render(board);

    // Human move.
    int move;
    while (true)
    {
        Console.Write("Your move (1-9): ");
        var input = Console.ReadLine();
        if (input is null)
        {
            Console.WriteLine("Goodbye!");
            return;
        }

        if (int.TryParse(input, out move) && move >= 1 && move <= 9 && board.IsEmpty(move - 1))
        {
            break;
        }

        Console.WriteLine("Invalid move, try again.");
    }

    board.Place(move - 1, Player.X);

    if (board.IsGameOver())
    {
        break;
    }

    // Computer move.
    var computerMove = computer.ChooseMove(board);
    board.Place(computerMove, Player.O);
    Console.WriteLine($"The computer plays cell {computerMove + 1}.");
}

Render(board);

var winner = board.Winner();
Console.WriteLine(winner switch
{
    Player.X => "You win!",
    Player.O => "The computer wins!",
    _ => "It's a draw!",
});

static void Render(Board board)
{
    Console.WriteLine();
    for (var row = 0; row < 3; row++)
    {
        var cells = new string[3];
        for (var col = 0; col < 3; col++)
        {
            var index = row * 3 + col;
            cells[col] = board.GetCell(index) switch
            {
                Player.X => "X",
                Player.O => "O",
                _ => (index + 1).ToString(),
            };
        }

        Console.WriteLine($" {cells[0]} | {cells[1]} | {cells[2]} ");
        if (row < 2)
        {
            Console.WriteLine("---+---+---");
        }
    }

    Console.WriteLine();
}
```

```csharp
/// next to the running program.

public class HighScores
{
    private const string FilePath = "highscores.json";

    public int HumanWins { get; set; }

    public int ComputerWins { get; set; }

    public int Draws { get; set; }

    public static HighScores Load()
    {
        if (!File.Exists(FilePath))
        {
            return new HighScores();
        }

        var json = File.ReadAllText(FilePath);
        return JsonConvert.DeserializeObject<HighScores>(json) ?? new HighScores();
    }

    public void Save()
    {
        var json = JsonConvert.SerializeObject(this, Formatting.Indented);
        File.WriteAllText(FilePath, json);
    }
}
```

````markdown
## Questions

1. *Why did you not have to touch any BUILD file in this exercise?*
   <details><summary>Answer</summary>
   Gazelle regenerates the per-directory BUILD files from the source `using`
   directives and the lock file; those BUILD files are derived artifacts, not
   source.
   </details>
2. *What would happen if you repinned but forgot to rerun Gazelle?*
   <details><summary>Answer</summary>
   The build still fails: the package is fetchable, but no target depends on
   it (and `HighScores.cs` is in no target). Resolution and code generation
   are separate steps.
   </details>
3. *Which files in this repository are hand-written, and which are generated
   or pinned?*
   <details><summary>Answer</summary>
   Hand-written: the C# sources, `MODULE.bazel`, and the root `BUILD.bazel`
   scaffolding (gazelle targets, alias, directive). Generated by Gazelle: the
   per-directory BUILD files. Generated but checked in (pinned):
   `nuget_resolve.json`.
   </details>
````

Create `/home/jhobert/Projects/Gazelle/dotnet_workshop/scripts/ci.sh` with exactly:

```bash
#!/usr/bin/env bash
# CI verification for the dotnet workshop repository.
#
# For every exercise branch (in a scratch clone):
#   1. Exercises 000, 002 and 004 must FAIL to build in their starting state
#      (the didactic motivation).
#   2. Apply the reference-solution patch from docs/solutions/ (kept on main).
#   3. bazel build //... must succeed with the solution applied.
#   4. Exercise 001: piped-input smoke run; a win/draw banner must appear.
#      (From exercise 002 onward the Spectre.Console SelectionPrompt needs an
#      interactive terminal, so those branches are verified with builds only.)
#   5. Exercises 003 and 004 solutions: `bazel run //:gazelle` must be
#      idempotent (no diff afterwards).
#   6. Exercise 004 solution: `REPIN=1 bazel run @nuget.net10.0//:pin` must
#      leave the lock file unchanged (freshness).
# Finally: next_exercise.sh must walk the full branch sequence, discarding
# uncommitted state at every hop, and print the completion message at the end.
#
# Local runs on restricted machines may need:
#   export JAVA_HOME=<path to a readable JDK>
#   BAZEL_GAZELLE_FLAGS="--@rules_go//go/config:pure"   # if cgo headers are unreadable
set -euo pipefail

REPO_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
SOLUTIONS="$REPO_ROOT/docs/solutions"
WORK="$(mktemp -d)"
BAZEL_GAZELLE_FLAGS="${BAZEL_GAZELLE_FLAGS:-}"

BROKEN_START_BRANCHES=(
    "exercise-000-intro"
    "exercise-002-nuget-resolve"
    "exercise-004-change"
)
GAZELLE_BRANCHES=(
    "exercise-003-gazelle"
    "exercise-004-change"
)
ALL_EXERCISES=(
    "exercise-000-intro"
    "exercise-001-hello-bazel"
    "exercise-002-nuget-resolve"
    "exercise-003-gazelle"
    "exercise-004-change"
)

contains() {
    local needle="$1"
    shift
    local item
    for item in "$@"; do
        if [[ "$item" == "$needle" ]]; then
            return 0
        fi
    done
    return 1
}

fail() {
    echo "CI FAILED: $*" >&2
    exit 1
}

git clone -q "$REPO_ROOT" "$WORK/repo"
cd "$WORK/repo"

for branch in "${ALL_EXERCISES[@]}"; do
    echo "=== $branch ==="
    git checkout -q "$branch"

    # Word splitting of $extra_flags is intentional (it holds 0 or 1 flags).
    extra_flags=""
    if contains "$branch" "${GAZELLE_BRANCHES[@]}"; then
        extra_flags="$BAZEL_GAZELLE_FLAGS"
    fi

    if contains "$branch" "${BROKEN_START_BRANCHES[@]}"; then
        if bazel build //... $extra_flags > /dev/null 2>&1; then
            fail "$branch: starting state should NOT build"
        fi
        echo "starting state fails to build (expected)"
    fi

    patch="$SOLUTIONS/$branch.patch"
    if [[ -f "$patch" ]]; then
        git apply "$patch"
        git add -A
        bazel build //... $extra_flags
        echo "reference solution builds"
    fi

    if [[ "$branch" == "exercise-001-hello-bazel" ]]; then
        smoke_out=""
        if ! smoke_out="$(printf '1\n2\n3\n4\n5\n6\n7\n8\n9\n' | bazel run //:app 2>&1)"; then
            echo "$smoke_out"
            fail "$branch: smoke run command itself failed"
        fi
        if ! grep -qE 'wins!|draw!' <<< "$smoke_out"; then
            echo "$smoke_out"
            fail "$branch: smoke run did not print a win/draw banner"
        fi
        echo "smoke run printed a win/draw banner"
    fi

    if contains "$branch" "${GAZELLE_BRANCHES[@]}"; then
        bazel run //:gazelle $extra_flags
        git diff --exit-code || fail "$branch: gazelle is not idempotent"
        echo "gazelle idempotent"
    fi

    if [[ "$branch" == "exercise-004-change" ]]; then
        REPIN=1 bazel run @nuget.net10.0//:pin
        git diff --exit-code || fail "$branch: lock file is stale"
        echo "lock file fresh"
    fi
done

echo "=== next_exercise.sh sequence ==="
git clone -q "$REPO_ROOT" "$WORK/sequence"
cd "$WORK/sequence"
git checkout -q main

expect_next() {
    local expected="$1"
    echo "dirty" >> README.md
    echo "junk" > junk.txt
    echo yes | ./next_exercise.sh > /dev/null
    local landed
    landed="$(git rev-parse --abbrev-ref HEAD)"
    [[ "$landed" == "$expected" ]] \
        || fail "next_exercise.sh landed on $landed, expected $expected"
    [[ -z "$(git status --porcelain)" ]] \
        || fail "next_exercise.sh left a dirty tree on $expected"
    echo "arrived at $expected with a clean tree"
}

expect_next exercise-000-intro
expect_next exercise-001-hello-bazel
expect_next exercise-002-nuget-resolve
expect_next exercise-003-gazelle
expect_next exercise-004-change

complete_out="$(echo yes | ./next_exercise.sh)"
grep -q "Workshop complete" <<< "$complete_out" \
    || fail "next_exercise.sh did not print the completion message"
[[ "$(git rev-parse --abbrev-ref HEAD)" == "exercise-004-change" ]] \
    || fail "next_exercise.sh moved away from the last exercise"
echo "workshop completion message printed"

echo "CI PASSED"
```
